cmake_minimum_required(VERSION 3.15)
# macOS builds are *always* universal for Apple Silicon + Intel, min 12.0.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Universal Apple Silicon + Intel" FORCE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()
# On MSVC use the static C runtime: release EXE runs without VC++ redist.
# CMake policy CMP0091 must be enabled before project().
# /MT and /MTd: include MSVC runtime in the binary, no separate VC redist needed.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(a9tool VERSION 1.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(a9tool src/a9tool.c src/a9_lzss.c src/a9_md5.c src/a9_csv_validator.c src/csvp_schema.c src/a9_compare.c src/a9_port.c)
if(MSVC)
    target_compile_options(a9tool PRIVATE /W4 /utf-8)
else()
    target_compile_options(a9tool PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS a9tool RUNTIME DESTINATION bin)
enable_testing()
add_executable(a9_md5_test tests/md5_known.c src/a9_md5.c)
target_include_directories(a9_md5_test PRIVATE src)
add_test(NAME md5_known_vectors COMMAND a9_md5_test)
if(NOT WIN32)
add_test(NAME archive_roundtrip COMMAND ${CMAKE_COMMAND} -E env A9TOOL_BIN=$<TARGET_FILE:a9tool> ${CMAKE_CURRENT_SOURCE_DIR}/tests/roundtrip.sh)
add_test(NAME csv_manual_validation COMMAND ${CMAKE_COMMAND} -E env A9TOOL_BIN=$<TARGET_FILE:a9tool> ${CMAKE_CURRENT_SOURCE_DIR}/tests/csv_validation.sh)
add_test(NAME archive_compare COMMAND ${CMAKE_COMMAND} -E env A9TOOL_BIN=$<TARGET_FILE:a9tool> ${CMAKE_CURRENT_SOURCE_DIR}/tests/compare.sh)
endif()
